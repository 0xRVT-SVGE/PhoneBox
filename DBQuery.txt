-- ============================
-- STUDENTS TABLE
-- ============================
CREATE TABLE students (
    sid CHAR(5) PRIMARY KEY
        CHECK (sid ~ '^E[0-9]{4}$' AND char_length(sid) = 5),
    last_name VARCHAR(40) NOT NULL,
    first_name VARCHAR(80) NOT NULL,
    embed FLOAT8[] NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    modified_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================
-- LOCATIONS TABLE
-- ============================
CREATE TABLE locations (
    lid SERIAL PRIMARY KEY,
    x SMALLINT NOT NULL CHECK (x BETWEEN 1 AND 500),
    y SMALLINT NOT NULL CHECK (y BETWEEN 1 AND 500),
    UNIQUE (x, y)
);

CREATE TABLE location_cursor (
    id BOOLEAN PRIMARY KEY DEFAULT TRUE,
    x SMALLINT NOT NULL DEFAULT 1,
    y SMALLINT NOT NULL DEFAULT 1
);

INSERT INTO location_cursor DEFAULT VALUES;

CREATE TABLE free_locations (
    lid INT PRIMARY KEY
        REFERENCES locations(lid)
        ON DELETE CASCADE
);


-- ============================
-- PHONES TABLE
-- ============================
CREATE TABLE phones (
    pid UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sid CHAR(5) NOT NULL
        REFERENCES students(sid)
        ON UPDATE CASCADE
        ON DELETE CASCADE,
    model VARCHAR(50) NOT NULL,
    imei VARCHAR(20) UNIQUE NOT NULL,
    cond VARCHAR(30)
        CHECK (cond IN ('New', 'Good', 'Fair', 'Damaged', 'Broken') OR cond IS NULL),
    admin_note TEXT,
    stud_note TEXT,
    is_stored BOOLEAN DEFAULT FALSE,
    lid INT UNIQUE NOT NULL
        REFERENCES locations(lid)
        ON DELETE CASCADE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    modified_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================
-- FUNCTION: Update modified_at
-- ============================
CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER AS $$
BEGIN
   NEW.modified_at = NOW();
   RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================
-- FUNCTION: Assign free location
-- ============================
CREATE OR REPLACE FUNCTION assign_or_create_location(x_max INT, y_max INT)
RETURNS INT AS $$
DECLARE
    reuse_lid INT;
    cur_x SMALLINT;
    cur_y SMALLINT;
    new_lid INT;
BEGIN
    -- Try reuse first
    SELECT lid
    INTO reuse_lid
    FROM free_locations
    ORDER BY lid
    LIMIT 1
    FOR UPDATE SKIP LOCKED;

    IF reuse_lid IS NOT NULL THEN
        DELETE FROM free_locations WHERE lid = reuse_lid;
        RETURN reuse_lid;
    END IF;

    -- Fallback to cursor (same as before)
    SELECT x, y INTO cur_x, cur_y
    FROM location_cursor
    FOR UPDATE;

    INSERT INTO locations (x, y)
    VALUES (cur_x, cur_y)
    RETURNING lid INTO new_lid;

    cur_y := cur_y + 1;
    IF cur_y > y_max THEN
        cur_y := 1;
        cur_x := cur_x + 1;
    END IF;

    IF cur_x > x_max THEN
        RAISE EXCEPTION 'No more available locations';
    END IF;

    UPDATE location_cursor SET x = cur_x, y = cur_y;

    RETURN new_lid;
END;
$$ LANGUAGE plpgsql;

-- ============================
-- FUNCTION: Auto-assign phone location
-- ============================
CREATE OR REPLACE FUNCTION auto_assign_location()
RETURNS TRIGGER AS $$
BEGIN
    NEW.lid := assign_or_create_location(500, 500);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================
-- FUNCTION: Generate the UUID
-- ============================
CREATE OR REPLACE FUNCTION phones_defaults()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.pid IS NULL THEN
        NEW.pid := gen_random_uuid();
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================
-- FUNCTION: Mark free location
-- ============================
CREATE OR REPLACE FUNCTION phones_after_delete()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO free_locations (lid)
    VALUES (OLD.lid)
    ON CONFLICT DO NOTHING;

    RETURN OLD;
END;
$$ LANGUAGE plpgsql;


-- ============================
-- TRIGGERS
-- ============================

-- Auto-assign unique location
CREATE TRIGGER phones_before_insert_location
BEFORE INSERT ON phones
FOR EACH ROW
EXECUTE FUNCTION auto_assign_location();

-- Generate the UUID
CREATE TRIGGER phones_before_insert_defaults
BEFORE INSERT ON phones
FOR EACH ROW
EXECUTE FUNCTION phones_defaults();

-- Mark free location
CREATE TRIGGER phones_after_delete_trigger
AFTER DELETE ON phones
FOR EACH ROW
EXECUTE FUNCTION phones_after_delete();

-- Update modified_at when student changes
CREATE TRIGGER students_before_update
BEFORE UPDATE ON students
FOR EACH ROW
EXECUTE FUNCTION update_modified_column();

-- Update modified_at when embedding changes
CREATE TRIGGER students_embed_modified
BEFORE UPDATE ON students
FOR EACH ROW
EXECUTE FUNCTION update_modified_column();

-- Update modified_at when phone changes
CREATE TRIGGER phones_before_update
BEFORE UPDATE ON phones
FOR EACH ROW
EXECUTE FUNCTION update_modified_column();
