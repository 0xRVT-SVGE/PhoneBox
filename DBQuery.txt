CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pg_trgm;

-- ── BOXES ─────────────────────────────────────────────────────────────────────
CREATE TABLE boxes (
    box_id             SERIAL       PRIMARY KEY,
    box_slug           TEXT         NOT NULL UNIQUE,
    box_name           TEXT         NOT NULL,
    max_slots          INT          NOT NULL DEFAULT 30,
    accept_year_groups SMALLINT[]   DEFAULT NULL,
    created_at         TIMESTAMPTZ  DEFAULT NOW()
);
COMMENT ON TABLE  boxes          IS 'Registry of physical phone-storage cabinets.';
COMMENT ON COLUMN boxes.box_slug IS 'Matches PHONEBOX_BOX_SLUG env var on the server.';
COMMENT ON COLUMN boxes.accept_year_groups IS 'NULL=any student. {1}=year 1 only.';

INSERT INTO boxes (box_id, box_slug, box_name, max_slots, accept_year_groups)

-- ── STUDENTS ──────────────────────────────────────────────────────────────────
CREATE TABLE students (
    sid         CHAR(5)      PRIMARY KEY
                             CHECK (sid ~ '^E[0-9]{4}$' AND char_length(sid) = 5),
    last_name   VARCHAR(40)  NOT NULL,
    first_name  VARCHAR(80)  NOT NULL,
    embed       FLOAT8[]     NOT NULL,
    year_group  SMALLINT     DEFAULT NULL
                             CHECK (year_group IS NULL OR year_group BETWEEN 1 AND 10),
    created_at  TIMESTAMPTZ  DEFAULT NOW(),
    modified_at TIMESTAMPTZ  DEFAULT NOW()
);
COMMENT ON TABLE  students            IS 'Registered students with face embedding.';
COMMENT ON COLUMN students.year_group IS 'Academic year group (1-10). NULL = unrestricted.';

-- Opt #36b trigram GIN indexes
CREATE INDEX idx_students_fname_trgm ON students USING gin(first_name gin_trgm_ops);
CREATE INDEX idx_students_lname_trgm ON students USING gin(last_name  gin_trgm_ops);

-- ── LOCATIONS ─────────────────────────────────────────────────────────────────
CREATE TABLE locations (
    lid    SERIAL   PRIMARY KEY,
    box_id INT      NOT NULL DEFAULT 1 REFERENCES boxes(box_id),
    x      SMALLINT NOT NULL CHECK (x BETWEEN 1 AND 500),
    y      SMALLINT NOT NULL CHECK (y BETWEEN 1 AND 500),
    UNIQUE (box_id, x, y)
);
ALTER SEQUENCE locations_lid_seq MINVALUE 0 START 0 RESTART 0;
COMMENT ON TABLE  locations        IS 'Static physical slot grid, scoped per box.';
COMMENT ON COLUMN locations.box_id IS 'Which physical cabinet this slot belongs to.';

CREATE INDEX idx_locations_box ON locations(box_id);

-- ── PHONES ────────────────────────────────────────────────────────────────────
CREATE TABLE phones (
    pid               UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
    sid               CHAR(5)     NOT NULL REFERENCES students(sid)
                                  ON UPDATE CASCADE ON DELETE CASCADE,
    model             VARCHAR(50) NOT NULL,
    imei              VARCHAR(20) UNIQUE NOT NULL,
    cond              VARCHAR(30) CHECK (cond IN ('New','Good','Fair','Damaged','Broken') OR cond IS NULL),
    phone_size        TEXT        NOT NULL DEFAULT 'standard'
                                  CHECK (phone_size IN ('standard','large')),
    allowed_box_slugs TEXT[]      DEFAULT NULL,
    admin_note        TEXT,
    stud_note         TEXT,
    created_at        TIMESTAMPTZ DEFAULT NOW(),
    modified_at       TIMESTAMPTZ DEFAULT NOW()
);
COMMENT ON TABLE  phones                   IS 'Student phones (not tied to a slot directly).';
COMMENT ON COLUMN phones.phone_size        IS '''standard'' or ''large''. Set at registration.';
COMMENT ON COLUMN phones.allowed_box_slugs IS 'NULL=any box. Non-null=deposit rejected elsewhere.';

CREATE INDEX idx_phones_sid           ON phones(sid);
CREATE INDEX idx_phones_imei          ON phones(imei);
CREATE INDEX idx_phones_allowed_slugs ON phones USING GIN(allowed_box_slugs);

-- ── PHONE STORAGE ─────────────────────────────────────────────────────────────
CREATE TABLE phone_storage (
    id           SERIAL      PRIMARY KEY,
    pid          UUID        NOT NULL REFERENCES phones(pid)    ON DELETE CASCADE,
    lid          INT         NOT NULL REFERENCES locations(lid) ON DELETE CASCADE,
    box_id       INT         REFERENCES boxes(box_id),
    stored_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    retrieved_at TIMESTAMPTZ,
    CONSTRAINT chk_retrieval_after_storage
        CHECK (retrieved_at IS NULL OR retrieved_at > stored_at)
);
COMMENT ON TABLE  phone_storage              IS 'Active + historical slot assignments. 1-month retention on retrieved rows.';
COMMENT ON COLUMN phone_storage.retrieved_at IS 'NULL=currently stored. NOT NULL=retrieved.';
COMMENT ON COLUMN phone_storage.lid          IS 'Auto-assigned to first empty slot in the box when NULL on INSERT.';
COMMENT ON COLUMN phone_storage.box_id       IS 'Denormalised from locations.box_id at deposit time.';

-- Opt #35 partial unique indexes
CREATE UNIQUE INDEX uniq_active_phone_storage ON phone_storage(pid) WHERE retrieved_at IS NULL;
CREATE UNIQUE INDEX uniq_active_location      ON phone_storage(lid) WHERE retrieved_at IS NULL;
COMMENT ON INDEX uniq_active_phone_storage IS 'Opt #35: prevents double-deposit of same phone.';
COMMENT ON INDEX uniq_active_location      IS 'Opt #35: prevents two phones in one slot.';

CREATE INDEX idx_phone_storage_pid        ON phone_storage(pid);
CREATE INDEX idx_phone_storage_lid        ON phone_storage(lid);
CREATE INDEX idx_phone_storage_active     ON phone_storage(pid, lid)  WHERE retrieved_at IS NULL;
CREATE INDEX idx_phone_storage_retrieved  ON phone_storage(retrieved_at) WHERE retrieved_at IS NOT NULL;
CREATE INDEX idx_phone_storage_box        ON phone_storage(box_id);
CREATE INDEX idx_phone_storage_box_active ON phone_storage(box_id)    WHERE retrieved_at IS NULL;
-- Opt #36c date-range scans
CREATE INDEX idx_phone_storage_stored_at  ON phone_storage(stored_at);

-- ── PHONE OPERATIONS AUDIT LOG ────────────────────────────────────────────────
CREATE TABLE phone_operations (
    id        SERIAL      PRIMARY KEY,
    pid       UUID        NOT NULL REFERENCES phones(pid) ON DELETE CASCADE,
    operation VARCHAR(20) NOT NULL CHECK (operation IN ('INSERT','UPDATE','DELETE')),
    old_data  JSONB,
    new_data  JSONB,
    timestamp TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
COMMENT ON TABLE phone_operations IS 'Audit log for phones table. 1-month retention.';
CREATE INDEX idx_phone_operations_pid_ts ON phone_operations(pid, timestamp DESC);
CREATE INDEX idx_phone_operations_ts     ON phone_operations(timestamp DESC);

-- ── SLOT BASELINES ────────────────────────────────────────────────────────────
CREATE TABLE slot_baselines (
    lid           INT         PRIMARY KEY REFERENCES locations(lid) ON DELETE CASCADE,
    box_id        INT         NOT NULL DEFAULT 1 REFERENCES boxes(box_id),
    embedding     BYTEA       NOT NULL,
    embedding_vec vector(96),
    calibrated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
COMMENT ON TABLE  slot_baselines               IS 'Per-slot visual embedding. embedding=BYTEA always written; embedding_vec=vector(96) written by Python.';
COMMENT ON COLUMN slot_baselines.box_id        IS 'Denormalised from locations.box_id for fast per-box baseline queries.';
COMMENT ON COLUMN slot_baselines.embedding_vec IS 'Opt #36a vector. NULL until Python writes it. HNSW index covers only non-NULL rows.';

CREATE INDEX idx_slot_baselines_box ON slot_baselines(box_id);
-- Opt #36a HNSW ANN index
CREATE INDEX idx_slot_baselines_hnsw
    ON slot_baselines USING hnsw (embedding_vec vector_cosine_ops)
    WITH (m = 16, ef_construction = 64)
    WHERE embedding_vec IS NOT NULL;
COMMENT ON INDEX idx_slot_baselines_hnsw IS 'Opt #36a: HNSW cosine ANN on 96-dim embeddings.';

-- ── EVIDENCE SYSTEM ───────────────────────────────────────────────────────────
CREATE TABLE evidence_sessions (

COMMENT ON TABLE evidence_sessions IS 'One row per admin resolution session.';

COMMENT ON TABLE  evidence_items     IS 'One row per video clip / event within a session.';

-- ── FUNCTIONS ─────────────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN NEW.modified_at = NOW(); RETURN NEW; END;
$$;

CREATE OR REPLACE FUNCTION phones_defaults()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.pid IS NULL THEN NEW.pid := gen_random_uuid(); END IF;
    RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION log_phone_operation()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE op_type VARCHAR(20); old_json JSONB; new_json JSONB;
BEGIN
    CASE TG_OP
        WHEN 'INSERT' THEN op_type := 'INSERT'; old_json := NULL;          new_json := to_jsonb(NEW);
        WHEN 'UPDATE' THEN op_type := 'UPDATE'; old_json := to_jsonb(OLD); new_json := to_jsonb(NEW);
        WHEN 'DELETE' THEN op_type := 'DELETE'; old_json := to_jsonb(OLD); new_json := NULL;
    END CASE;
    INSERT INTO phone_operations(pid, operation, old_data, new_data, timestamp)
    VALUES (COALESCE(NEW.pid, OLD.pid), op_type, old_json, new_json, NOW());
    RETURN CASE WHEN TG_OP = 'DELETE' THEN OLD ELSE NEW END;
END;
$$;

-- Multi-box-aware auto slot assignment (safety net; Python always passes lid)
CREATE OR REPLACE FUNCTION auto_assign_location()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.lid IS NULL THEN
        IF NEW.box_id IS NULL THEN
            RAISE EXCEPTION 'phone_storage: cannot auto-assign lid — box_id must be set';
        END IF;
        SELECT l.lid INTO NEW.lid
        FROM   locations l
        WHERE  l.box_id = NEW.box_id
          AND  NOT EXISTS (
                   SELECT 1 FROM phone_storage ps
                   WHERE ps.lid = l.lid AND ps.retrieved_at IS NULL
               )
        ORDER BY l.lid LIMIT 1
        FOR UPDATE SKIP LOCKED;
        IF NEW.lid IS NULL THEN
            RAISE EXCEPTION 'No empty locations available in box_id=%', NEW.box_id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$;
COMMENT ON FUNCTION auto_assign_location() IS 'Assigns first free slot in NEW.box_id when NEW.lid IS NULL. Multi-box aware.';

-- Opt #25 LISTEN/NOTIFY
CREATE OR REPLACE FUNCTION notify_storage_change()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    PERFORM pg_notify(
        'phonebox_storage_change',
        CASE WHEN TG_OP = 'DELETE' THEN OLD.lid::text ELSE NEW.lid::text END
    );
    RETURN NULL;
END;
$$;

CREATE OR REPLACE FUNCTION cleanup_old_logs()
RETURNS TABLE(deleted_phone_ops INT, deleted_storage INT)
LANGUAGE plpgsql AS $$
DECLARE d_ops INT; d_stor INT;
BEGIN
    DELETE FROM phone_operations WHERE timestamp    < NOW() - INTERVAL '1 month';
    GET DIAGNOSTICS d_ops  = ROW_COUNT;
    DELETE FROM phone_storage   WHERE retrieved_at IS NOT NULL
                                  AND retrieved_at < NOW() - INTERVAL '1 month';
    GET DIAGNOSTICS d_stor = ROW_COUNT;
    deleted_phone_ops := d_ops; deleted_storage := d_stor; RETURN NEXT;
    RAISE NOTICE 'Cleanup: % phone_ops + % storage rows deleted', d_ops, d_stor;
END;
$$;

-- ── TRIGGERS ──────────────────────────────────────────────────────────────────

CREATE TRIGGER phones_before_insert_defaults
    BEFORE INSERT ON phones FOR EACH ROW EXECUTE FUNCTION phones_defaults();

CREATE TRIGGER phones_before_update
    BEFORE UPDATE ON phones FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER phones_after_all_operations
    AFTER INSERT OR UPDATE OR DELETE ON phones FOR EACH ROW EXECUTE FUNCTION log_phone_operation();

CREATE TRIGGER students_before_update
    BEFORE UPDATE ON students FOR EACH ROW EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER phone_storage_before_insert_auto_lid
    BEFORE INSERT ON phone_storage FOR EACH ROW EXECUTE FUNCTION auto_assign_location();

CREATE TRIGGER phone_storage_notify_change
    AFTER INSERT OR UPDATE OR DELETE ON phone_storage FOR EACH ROW EXECUTE FUNCTION notify_storage_change();
COMMENT ON TRIGGER phone_storage_notify_change ON phone_storage IS
    'Opt #25: pg_notify("phonebox_storage_change", lid) — Python asyncpg invalidates pid cache in <1 ms.';

-- ── VIEWS ─────────────────────────────────────────────────────────────────────

CREATE VIEW v_active_phone_storage AS
SELECT
    ps.id, ps.pid, ps.lid, ps.stored_at,
    ps.box_id, b.box_slug, b.box_name,
    p.sid, p.imei, p.model, p.cond, p.phone_size, p.allowed_box_slugs,
    l.x, l.y,
    st.first_name, st.last_name, st.year_group
FROM phone_storage ps
JOIN phones    p  ON ps.pid    = p.pid
JOIN locations l  ON ps.lid    = l.lid
JOIN boxes     b  ON ps.box_id = b.box_id
JOIN students  st ON p.sid     = st.sid
WHERE ps.retrieved_at IS NULL;
COMMENT ON VIEW v_active_phone_storage IS 'All phones currently in storage with full student, box, and location context.';

CREATE VIEW v_empty_locations AS
SELECT l.lid, l.x, l.y, l.box_id, b.box_slug, b.box_name
FROM   locations l
JOIN   boxes b ON l.box_id = b.box_id
LEFT JOIN phone_storage ps ON l.lid = ps.lid AND ps.retrieved_at IS NULL
WHERE  ps.lid IS NULL;
COMMENT ON VIEW v_empty_locations IS 'All unoccupied slots with box context.';

CREATE VIEW v_box_dashboard AS
SELECT
    b.box_id, b.box_slug, b.box_name, b.max_slots, b.accept_year_groups,
    COUNT(ps.id)                                        AS phones_stored,
    b.max_slots - COUNT(ps.id)                          AS slots_free,
    ROUND(COUNT(ps.id)::NUMERIC / b.max_slots * 100, 1) AS fill_pct
FROM boxes b
LEFT JOIN locations     l  ON l.box_id = b.box_id
LEFT JOIN phone_storage ps ON ps.lid   = l.lid AND ps.retrieved_at IS NULL
GROUP BY b.box_id, b.box_slug, b.box_name, b.max_slots, b.accept_year_groups
ORDER BY b.box_id;
COMMENT ON VIEW v_box_dashboard IS 'Cross-box fill summary — used by central dashboard endpoint.';

-- ── HELPER FUNCTIONS ──────────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION get_phone_storage_history(phone_pid UUID, months_back INT DEFAULT 1)
RETURNS TABLE(storage_id INT, lid INT, x SMALLINT, y SMALLINT,
              stored_at TIMESTAMPTZ, retrieved_at TIMESTAMPTZ, duration INTERVAL)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT ps.id, ps.lid, l.x, l.y, ps.stored_at, ps.retrieved_at,
           COALESCE(ps.retrieved_at, NOW()) - ps.stored_at
    FROM   phone_storage ps
    JOIN   locations l ON ps.lid = l.lid
    WHERE  ps.pid = phone_pid
      AND  ps.stored_at > NOW() - (months_back || ' months')::INTERVAL
    ORDER  BY ps.stored_at DESC;
END;
$$;

CREATE OR REPLACE FUNCTION get_location_usage_history(location_lid INT, months_back INT DEFAULT 1)
RETURNS TABLE(phone_pid UUID, imei VARCHAR, student_sid CHAR,
              stored_at TIMESTAMPTZ, retrieved_at TIMESTAMPTZ, duration INTERVAL)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT ps.pid, p.imei, p.sid, ps.stored_at, ps.retrieved_at,
           COALESCE(ps.retrieved_at, NOW()) - ps.stored_at
    FROM   phone_storage ps
    JOIN   phones p ON ps.pid = p.pid
    WHERE  ps.lid = location_lid
      AND  ps.stored_at > NOW() - (months_back || ' months')::INTERVAL
    ORDER  BY ps.stored_at DESC;
END;
$$;

CREATE OR REPLACE FUNCTION get_available_locations(
    limit_count INT DEFAULT 10,
    for_box_id  INT DEFAULT NULL
)
RETURNS TABLE(lid INT, x SMALLINT, y SMALLINT)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT l.lid, l.x, l.y
    FROM   locations l
    LEFT JOIN phone_storage ps ON l.lid = ps.lid AND ps.retrieved_at IS NULL
    WHERE  ps.lid IS NULL
      AND (for_box_id IS NULL OR l.box_id = for_box_id)
    ORDER  BY l.lid
    LIMIT  limit_count;
END;
$$;

CREATE OR REPLACE FUNCTION get_box_lids(for_box_id INT)
RETURNS TABLE(lid INT, x SMALLINT, y SMALLINT)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT l.lid, l.x, l.y FROM locations l
    WHERE  l.box_id = for_box_id ORDER BY l.lid;
END;
$$;
COMMENT ON FUNCTION get_box_lids(INT) IS
    'Returns all (lid,x,y) for a given box. Used by ROI calibration and slot monitor startup.';
