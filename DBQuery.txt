-- ============================================================
-- PhoneBox — Complete Database Schema  (single authoritative file)
-- Sessions 1-6 optimizations included.
--
-- Optimizations embedded:
--   Opt #25  LISTEN/NOTIFY trigger for Python cache invalidation
--   Opt #35  Partial unique indexes on phone_storage (active-record lookups)
--   Opt #36a pgvector vector(96) column on slot_baselines + HNSW index
--   Opt #36b pg_trgm GIN indexes on student names (fast ILIKE search)
--   Opt #36c stored_at index on phone_storage (activity-report range queries)
--
-- IMPORTANT — vector migration:
--   The embedding_vec column starts NULL. Python (db_interface.py) writes
--   both BYTEA and vector(96) on every save_baseline() call. There is NO
--   PL/pgSQL bytea→float conversion here — that approach produces wrong
--   IEEE-754 values. Migration of existing rows is done by Python:
--   run `recalibrate_all` or wait for the next natural baseline save.
--
-- Run order: top to bottom, once, against an empty PhoneBoxDB.
--   psql -d PhoneBoxDB -f DBQuery.sql
-- ============================================================


-- ============================================================
-- EXTENSIONS
-- ============================================================

-- Opt #36a: pgvector — native vector type + HNSW/IVFFlat ANN indexes
CREATE EXTENSION IF NOT EXISTS vector;

-- Opt #36b: pg_trgm — GIN trigram indexes for fast ILIKE '%query%'
CREATE EXTENSION IF NOT EXISTS pg_trgm;


-- ============================================================
-- STUDENTS
-- ============================================================

CREATE TABLE students (
    sid         CHAR(5)       PRIMARY KEY
                              CHECK (sid ~ '^E[0-9]{4}$' AND char_length(sid) = 5),
    last_name   VARCHAR(40)   NOT NULL,
    first_name  VARCHAR(80)   NOT NULL,
    embed       FLOAT8[]      NOT NULL,
    created_at  TIMESTAMPTZ   DEFAULT NOW(),
    modified_at TIMESTAMPTZ   DEFAULT NOW()
);

-- Opt #36b: trigram GIN indexes — turns seq-scan ILIKE into indexed lookup
CREATE INDEX idx_students_fname_trgm
    ON students USING gin(first_name gin_trgm_ops);
CREATE INDEX idx_students_lname_trgm
    ON students USING gin(last_name gin_trgm_ops);


-- ============================================================
-- LOCATIONS  (static physical slot grid)
-- ============================================================

CREATE TABLE locations (
    lid  SERIAL    PRIMARY KEY,
    x    SMALLINT  NOT NULL CHECK (x BETWEEN 1 AND 500),
    y    SMALLINT  NOT NULL CHECK (y BETWEEN 1 AND 500),
    UNIQUE (x, y)
);

-- Reset sequence to start at 0
ALTER SEQUENCE locations_lid_seq MINVALUE 0 START 0 RESTART 0;

-- ============================================================
-- PHONES
-- ============================================================

CREATE TABLE phones (
    pid         UUID          PRIMARY KEY DEFAULT gen_random_uuid(),
    sid         CHAR(5)       NOT NULL
                              REFERENCES students(sid)
                              ON UPDATE CASCADE ON DELETE CASCADE,
    model       VARCHAR(50)   NOT NULL,
    imei        VARCHAR(20)   UNIQUE NOT NULL,
    cond        VARCHAR(30)
                              CHECK (cond IN ('New','Good','Fair','Damaged','Broken')
                                     OR cond IS NULL),
    admin_note  TEXT,
    stud_note   TEXT,
    created_at  TIMESTAMPTZ   DEFAULT NOW(),
    modified_at TIMESTAMPTZ   DEFAULT NOW()
);

CREATE INDEX idx_phones_sid  ON phones(sid);
CREATE INDEX idx_phones_imei ON phones(imei);


-- ============================================================
-- PHONE STORAGE  (which phone is in which slot right now)
-- ============================================================

CREATE TABLE phone_storage (
    id           SERIAL       PRIMARY KEY,
    pid          UUID         NOT NULL REFERENCES phones(pid) ON DELETE CASCADE,
    lid          INT          NOT NULL REFERENCES locations(lid) ON DELETE CASCADE,
    stored_at    TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    retrieved_at TIMESTAMPTZ,
    CONSTRAINT chk_retrieval_after_storage
        CHECK (retrieved_at IS NULL OR retrieved_at > stored_at)
);

-- Opt #35: partial unique indexes — O(1) active-record constraint checks
-- These replace full-table scans on the hot deposit/withdraw paths.
CREATE UNIQUE INDEX uniq_active_phone_storage
    ON phone_storage(pid)
    WHERE retrieved_at IS NULL;

CREATE UNIQUE INDEX uniq_active_location
    ON phone_storage(lid)
    WHERE retrieved_at IS NULL;

-- General access patterns
CREATE INDEX idx_phone_storage_pid       ON phone_storage(pid);
CREATE INDEX idx_phone_storage_lid       ON phone_storage(lid);
CREATE INDEX idx_phone_storage_active    ON phone_storage(pid, lid)
    WHERE retrieved_at IS NULL;
CREATE INDEX idx_phone_storage_retrieved ON phone_storage(retrieved_at)
    WHERE retrieved_at IS NOT NULL;

-- Opt #36c: date-range scans for activity report (stored_at OR retrieved_at)
CREATE INDEX idx_phone_storage_stored_at ON phone_storage(stored_at);


-- ============================================================
-- PHONE OPERATION AUDIT LOG  (1 month retention)
-- ============================================================

CREATE TABLE phone_operations (
    id        SERIAL       PRIMARY KEY,
    pid       UUID         NOT NULL REFERENCES phones(pid) ON DELETE CASCADE,
    operation VARCHAR(20)  NOT NULL CHECK (operation IN ('INSERT','UPDATE','DELETE')),
    old_data  JSONB,
    new_data  JSONB,
    timestamp TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_phone_operations_pid_ts ON phone_operations(pid, timestamp DESC);
CREATE INDEX idx_phone_operations_ts     ON phone_operations(timestamp DESC);


-- ============================================================
-- SLOT BASELINES  (visual embeddings for each physical slot)
--
-- Dual-column design (Opt #36a):
--   embedding     BYTEA      — always written; used by all DB reads
--   embedding_vec vector(96) — written by Python when pgvector is active;
--                              starts NULL, fills in on first save_baseline()
--
-- The Python layer (db_interface.py) handles IEEE-754 byte conversion
-- correctly via numpy. No PL/pgSQL float decoding is attempted here.
-- ============================================================

CREATE TABLE slot_baselines (
    lid           INT         PRIMARY KEY REFERENCES locations(lid) ON DELETE CASCADE,
    embedding     BYTEA       NOT NULL,
    embedding_vec vector(96),           -- NULL until first post-migration save
    calibrated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Opt #36a: HNSW approximate-nearest-neighbour index for cosine search.
-- Partial: only indexes rows that have been migrated to vector(96).
-- m=16, ef_construction=64 are proven defaults for 96-dim embeddings.
CREATE INDEX idx_slot_baselines_hnsw
    ON slot_baselines USING hnsw (embedding_vec vector_cosine_ops)
    WITH (m = 16, ef_construction = 64)
    WHERE embedding_vec IS NOT NULL;


-- ============================================================
-- EVIDENCE SYSTEM
-- ============================================================

-- One row per admin resolution session
CREATE TABLE evidence_sessions (
    id         SERIAL        PRIMARY KEY,
    session_id VARCHAR(8)    NOT NULL UNIQUE,   -- secrets.token_hex(4)
    opened_at  TIMESTAMPTZ   NOT NULL,
    closed_at  TIMESTAMPTZ,
    -- Outcome values:
    --   'clean'       — all mismatches resolved, no anomalies
    --   'flagged'     — no_qr_found / declare_missing / warnings at close
    --   'force_closed'— session timed out or manually force-closed
    outcome    VARCHAR(20),
    warnings   JSONB,
    kept       BOOLEAN       NOT NULL DEFAULT FALSE
);

CREATE INDEX idx_evidence_sessions_kept
    ON evidence_sessions(kept)
    WHERE kept = TRUE;

-- One row per captured event / video clip within a session
CREATE TABLE evidence_items (
    id          SERIAL       PRIMARY KEY,
    session_id  VARCHAR(8)   NOT NULL
                             REFERENCES evidence_sessions(session_id)
                             ON DELETE CASCADE,
    captured_at TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    -- Event values: 'session_start','remove_phone','qr_scanned','no_qr_found',
    --   'stage_phone','unstage_phone','place_phone','declare_missing',
    --   'needs_deposit','session_close','video_clip'
    event_type  VARCHAR(50)  NOT NULL,
    lid         INTEGER,
    -- pid is VARCHAR not UUID: during admin resolution the pid may be
    -- 'pending-lid{N}' or 'unknown-{N}' before QR is confirmed.
    -- Application code uses the real UUID once confirmed.
    pid         VARCHAR(50),
    photo_path  VARCHAR(500),
    metadata    JSONB
);

CREATE INDEX idx_evidence_items_session ON evidence_items(session_id);
CREATE INDEX idx_evidence_items_event   ON evidence_items(event_type);


-- ============================================================
-- FUNCTIONS
-- ============================================================

-- Keep modified_at current on every UPDATE
CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    NEW.modified_at = NOW();
    RETURN NEW;
END;
$$;

-- Guarantee phones always get a UUID pid (safety net)
CREATE OR REPLACE FUNCTION phones_defaults()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.pid IS NULL THEN
        NEW.pid := gen_random_uuid();
    END IF;
    RETURN NEW;
END;
$$;

-- Append row to audit log on every phones change
CREATE OR REPLACE FUNCTION log_phone_operation()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
    op_type  VARCHAR(20);
    old_json JSONB;
    new_json JSONB;
BEGIN
    CASE TG_OP
        WHEN 'INSERT' THEN op_type := 'INSERT'; old_json := NULL;          new_json := to_jsonb(NEW);
        WHEN 'UPDATE' THEN op_type := 'UPDATE'; old_json := to_jsonb(OLD); new_json := to_jsonb(NEW);
        WHEN 'DELETE' THEN op_type := 'DELETE'; old_json := to_jsonb(OLD); new_json := NULL;
    END CASE;

    INSERT INTO phone_operations(pid, operation, old_data, new_data, timestamp)
    VALUES (COALESCE(NEW.pid, OLD.pid), op_type, old_json, new_json, NOW());

    RETURN CASE WHEN TG_OP = 'DELETE' THEN OLD ELSE NEW END;
END;
$$;

-- Auto-assign the first empty lid when lid IS NULL on INSERT
CREATE OR REPLACE FUNCTION auto_assign_location()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    IF NEW.lid IS NULL THEN
        SELECT l.lid INTO NEW.lid
        FROM   locations l
        WHERE  NOT EXISTS (
            SELECT 1 FROM phone_storage ps
            WHERE  ps.lid = l.lid AND ps.retrieved_at IS NULL
        )
        ORDER BY l.lid
        LIMIT 1;

        IF NEW.lid IS NULL THEN
            RAISE EXCEPTION 'No empty locations available for phone storage';
        END IF;
    END IF;
    RETURN NEW;
END;
$$;

-- Opt #25: broadcast lid via LISTEN/NOTIFY after every phone_storage change.
-- Python AsyncSlotMonitorDB listens on 'phonebox_storage_change' and
-- invalidates its in-process pid→lid cache immediately, without polling.
CREATE OR REPLACE FUNCTION notify_storage_change()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
    PERFORM pg_notify(
        'phonebox_storage_change',
        CASE WHEN TG_OP = 'DELETE' THEN OLD.lid::text ELSE NEW.lid::text END
    );
    RETURN NULL;   -- AFTER trigger; return value ignored
END;
$$;

-- Delete log rows and completed storage rows older than 1 month.
-- Schedule weekly via pg_cron or cron + psql.
CREATE OR REPLACE FUNCTION cleanup_old_logs()
RETURNS TABLE(deleted_phone_ops INT, deleted_storage INT)
LANGUAGE plpgsql AS $$
DECLARE
    d_ops  INT;
    d_stor INT;
BEGIN
    DELETE FROM phone_operations  WHERE timestamp    < NOW() - INTERVAL '1 month';
    GET DIAGNOSTICS d_ops  = ROW_COUNT;

    DELETE FROM phone_storage
    WHERE retrieved_at IS NOT NULL
      AND retrieved_at < NOW() - INTERVAL '1 month';
    GET DIAGNOSTICS d_stor = ROW_COUNT;

    deleted_phone_ops := d_ops;
    deleted_storage   := d_stor;
    RETURN NEXT;

    RAISE NOTICE 'Cleanup: % phone_ops + % storage rows deleted', d_ops, d_stor;
END;
$$;


-- ============================================================
-- TRIGGERS
-- ============================================================

-- phones: guarantee UUID pid before INSERT
CREATE TRIGGER phones_before_insert_defaults
    BEFORE INSERT ON phones
    FOR EACH ROW EXECUTE FUNCTION phones_defaults();

-- phones: keep modified_at current
CREATE TRIGGER phones_before_update
    BEFORE UPDATE ON phones
    FOR EACH ROW EXECUTE FUNCTION update_modified_column();

-- phones: write audit log
CREATE TRIGGER phones_after_all_operations
    AFTER INSERT OR UPDATE OR DELETE ON phones
    FOR EACH ROW EXECUTE FUNCTION log_phone_operation();

-- students: keep modified_at current
CREATE TRIGGER students_before_update
    BEFORE UPDATE ON students
    FOR EACH ROW EXECUTE FUNCTION update_modified_column();

-- phone_storage: auto-assign empty lid when lid IS NULL on INSERT
CREATE TRIGGER phone_storage_before_insert_auto_lid
    BEFORE INSERT ON phone_storage
    FOR EACH ROW EXECUTE FUNCTION auto_assign_location();

-- Opt #25: broadcast every phone_storage change for Python cache invalidation
-- FOR EACH ROW so the payload carries the specific lid that changed.
CREATE TRIGGER phone_storage_notify_change
    AFTER INSERT OR UPDATE OR DELETE ON phone_storage
    FOR EACH ROW EXECUTE FUNCTION notify_storage_change();


-- ============================================================
-- VIEWS
-- ============================================================

-- All phones currently in storage with full context
CREATE OR REPLACE VIEW v_active_phone_storage AS
SELECT
    ps.id, ps.pid, ps.lid, ps.stored_at,
    p.sid, p.imei, p.model, p.cond,
    l.x, l.y,
    st.first_name, st.last_name
FROM phone_storage ps
JOIN phones   p  ON ps.pid = p.pid
JOIN locations l ON ps.lid = l.lid
JOIN students st ON p.sid  = st.sid
WHERE ps.retrieved_at IS NULL;

-- All slots with no active deposit
CREATE OR REPLACE VIEW v_empty_locations AS
SELECT l.lid, l.x, l.y
FROM   locations l
LEFT JOIN phone_storage ps ON l.lid = ps.lid AND ps.retrieved_at IS NULL
WHERE  ps.lid IS NULL;


-- ============================================================
-- HELPER FUNCTIONS (used by Python layer)
-- ============================================================

-- Full storage history for one phone (most recent first)
CREATE OR REPLACE FUNCTION get_phone_storage_history(
    phone_pid   UUID,
    months_back INT DEFAULT 1
)
RETURNS TABLE(
    storage_id   INT,
    lid          INT,
    x            SMALLINT,
    y            SMALLINT,
    stored_at    TIMESTAMPTZ,
    retrieved_at TIMESTAMPTZ,
    duration     INTERVAL
)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT ps.id, ps.lid, l.x, l.y, ps.stored_at, ps.retrieved_at,
           COALESCE(ps.retrieved_at, NOW()) - ps.stored_at AS duration
    FROM   phone_storage ps
    JOIN   locations l ON ps.lid = l.lid
    WHERE  ps.pid       = phone_pid
      AND  ps.stored_at > NOW() - (months_back || ' months')::INTERVAL
    ORDER  BY ps.stored_at DESC;
END;
$$;

-- Full usage history for one physical slot (most recent first)
CREATE OR REPLACE FUNCTION get_location_usage_history(
    location_lid INT,
    months_back  INT DEFAULT 1
)
RETURNS TABLE(
    phone_pid    UUID,
    imei         VARCHAR,
    student_sid  CHAR,
    stored_at    TIMESTAMPTZ,
    retrieved_at TIMESTAMPTZ,
    duration     INTERVAL
)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT ps.pid, p.imei, p.sid, ps.stored_at, ps.retrieved_at,
           COALESCE(ps.retrieved_at, NOW()) - ps.stored_at AS duration
    FROM   phone_storage ps
    JOIN   phones p ON ps.pid = p.pid
    WHERE  ps.lid       = location_lid
      AND  ps.stored_at > NOW() - (months_back || ' months')::INTERVAL
    ORDER  BY ps.stored_at DESC;
END;
$$;

-- Return first N empty locations (ordered by lid)
CREATE OR REPLACE FUNCTION get_available_locations(limit_count INT DEFAULT 10)
RETURNS TABLE(lid INT, x SMALLINT, y SMALLINT)
LANGUAGE plpgsql AS $$
BEGIN
    RETURN QUERY
    SELECT l.lid, l.x, l.y
    FROM   locations l
    LEFT JOIN phone_storage ps ON l.lid = ps.lid AND ps.retrieved_at IS NULL
    WHERE  ps.lid IS NULL
    ORDER  BY l.lid
    LIMIT  limit_count;
END;
$$;
