-- ============================
-- STUDENTS TABLE
-- ============================
CREATE TABLE students (
    sid CHAR(5) PRIMARY KEY
        CHECK (sid ~ '^E[0-9]{4}$' AND char_length(sid) = 5),
    last_name VARCHAR(40) NOT NULL,
    first_name VARCHAR(80) NOT NULL,
    embed FLOAT8[] NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    modified_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================
-- LOCATIONS TABLE (STATIC GRID)
-- ============================
CREATE TABLE locations (
    lid SERIAL PRIMARY KEY,
    x SMALLINT NOT NULL CHECK (x BETWEEN 1 AND 500),
    y SMALLINT NOT NULL CHECK (y BETWEEN 1 AND 500),
    UNIQUE (x, y)
);

-- ============================
-- PHONES TABLE
-- ============================
CREATE TABLE phones (
    pid UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sid CHAR(5) NOT NULL
        REFERENCES students(sid)
        ON UPDATE CASCADE
        ON DELETE CASCADE,
    model VARCHAR(50) NOT NULL,
    imei VARCHAR(20) UNIQUE NOT NULL,
    cond VARCHAR(30)
        CHECK (cond IN ('New', 'Good', 'Fair', 'Damaged', 'Broken') OR cond IS NULL),
    admin_note TEXT,
    stud_note TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    modified_at TIMESTAMPTZ DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_phones_sid ON phones(sid);
CREATE INDEX idx_phones_imei ON phones(imei);

-- ============================
-- PHONE STORAGE (JUNCTION TABLE)
-- ============================
CREATE TABLE phone_storage (
    id SERIAL PRIMARY KEY,
    pid UUID NOT NULL
        REFERENCES phones(pid)
        ON DELETE CASCADE,
    lid INT NOT NULL
        REFERENCES locations(lid)
        ON DELETE CASCADE,
    stored_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
    retrieved_at TIMESTAMPTZ,

    -- Constraint: retrieved_at must be after stored_at
    CONSTRAINT check_retrieval_time CHECK (
        retrieved_at IS NULL OR retrieved_at > stored_at
    )
);

-- CRITICAL: Only one active storage per phone
CREATE UNIQUE INDEX uniq_active_phone_storage
ON phone_storage(pid)
WHERE retrieved_at IS NULL;

-- CRITICAL: Only one phone per location at a time
CREATE UNIQUE INDEX uniq_active_location
ON phone_storage(lid)
WHERE retrieved_at IS NULL;

-- Performance indexes
CREATE INDEX idx_phone_storage_pid ON phone_storage(pid);
CREATE INDEX idx_phone_storage_lid ON phone_storage(lid);
CREATE INDEX idx_phone_storage_active ON phone_storage(pid, lid) WHERE retrieved_at IS NULL;
CREATE INDEX idx_phone_storage_retrieved ON phone_storage(retrieved_at) WHERE retrieved_at IS NOT NULL;

-- ============================
-- PHONE OPERATION LOGS (1 MONTH RETENTION)
-- ============================
CREATE TABLE phone_operations (
    id SERIAL PRIMARY KEY,
    pid UUID NOT NULL
        REFERENCES phones(pid)
        ON DELETE CASCADE,
    operation VARCHAR(20) NOT NULL
        CHECK (operation IN ('INSERT', 'UPDATE', 'DELETE')),
    old_data JSONB,
    new_data JSONB,
    timestamp TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

CREATE INDEX idx_phone_operations_pid_ts ON phone_operations(pid, timestamp DESC);
CREATE INDEX idx_phone_operations_ts ON phone_operations(timestamp DESC);

-- ============================
-- SLOT MONITORING - BASELINES ONLY
-- ============================

-- Baseline embeddings for each location (runtime only, no state tracking)
CREATE TABLE slot_baselines (
    lid INT PRIMARY KEY
        REFERENCES locations(lid)
        ON DELETE CASCADE,
    embedding BYTEA NOT NULL,
    calibrated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- ============================
-- FUNCTIONS
-- ============================

CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER AS $$
BEGIN
   NEW.modified_at = NOW();
   RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION phones_defaults()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.pid IS NULL THEN
        NEW.pid := gen_random_uuid();
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION log_phone_operation()
RETURNS TRIGGER AS $$
DECLARE
    op_type VARCHAR(20);
    old_json JSONB;
    new_json JSONB;
BEGIN
    IF TG_OP = 'INSERT' THEN
        op_type := 'INSERT';
        old_json := NULL;
        new_json := to_jsonb(NEW);
    ELSIF TG_OP = 'UPDATE' THEN
        op_type := 'UPDATE';
        old_json := to_jsonb(OLD);
        new_json := to_jsonb(NEW);
    ELSIF TG_OP = 'DELETE' THEN
        op_type := 'DELETE';
        old_json := to_jsonb(OLD);
        new_json := NULL;
    END IF;

    INSERT INTO phone_operations (pid, operation, old_data, new_data, timestamp)
    VALUES (
        COALESCE(NEW.pid, OLD.pid),
        op_type,
        old_json,
        new_json,
        NOW()
    );

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    ELSE
        RETURN NEW;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- Auto-assign empty lid if not provided
CREATE OR REPLACE FUNCTION auto_assign_location()
RETURNS TRIGGER AS $$
BEGIN
    -- If lid is NULL, find first available empty location
    IF NEW.lid IS NULL THEN
        SELECT l.lid INTO NEW.lid
        FROM locations l
        WHERE NOT EXISTS (
            SELECT 1 FROM phone_storage ps
            WHERE ps.lid = l.lid
              AND ps.retrieved_at IS NULL
        )
        ORDER BY l.lid
        LIMIT 1;

        -- If no empty location found, raise error
        IF NEW.lid IS NULL THEN
            RAISE EXCEPTION 'No empty locations available for phone storage';
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION cleanup_old_logs()
RETURNS TABLE(
    deleted_phone_ops INT,
    deleted_storage INT
) AS $$
DECLARE
    d_phone_ops INT;
    d_storage INT;
BEGIN
    -- Delete phone operations older than 1 month
    DELETE FROM phone_operations
    WHERE timestamp < NOW() - INTERVAL '1 month';
    GET DIAGNOSTICS d_phone_ops = ROW_COUNT;

    -- Delete phone storage records older than 1 month (already retrieved)
    DELETE FROM phone_storage
    WHERE retrieved_at IS NOT NULL
      AND retrieved_at < NOW() - INTERVAL '1 month';
    GET DIAGNOSTICS d_storage = ROW_COUNT;

    deleted_phone_ops := d_phone_ops;
    deleted_storage := d_storage;

    RETURN NEXT;

    RAISE NOTICE 'Cleanup complete: % phone ops, % storage records deleted',
        d_phone_ops, d_storage;
END;
$$ LANGUAGE plpgsql;

-- ============================
-- TRIGGERS
-- ============================

CREATE TRIGGER phones_before_insert_defaults
BEFORE INSERT ON phones
FOR EACH ROW
EXECUTE FUNCTION phones_defaults();

CREATE TRIGGER phones_before_update
BEFORE UPDATE ON phones
FOR EACH ROW
EXECUTE FUNCTION update_modified_column();

CREATE TRIGGER phones_after_all_operations
AFTER INSERT OR UPDATE OR DELETE ON phones
FOR EACH ROW
EXECUTE FUNCTION log_phone_operation();

CREATE TRIGGER students_before_update
BEFORE UPDATE ON students
FOR EACH ROW
EXECUTE FUNCTION update_modified_column();

-- Auto-assign location trigger
CREATE TRIGGER phone_storage_before_insert_auto_lid
BEFORE INSERT ON phone_storage
FOR EACH ROW
EXECUTE FUNCTION auto_assign_location();

CREATE OR REPLACE VIEW v_active_phone_storage AS
SELECT
    ps.id,
    ps.pid,
    ps.lid,
    ps.stored_at,
    p.sid,
    p.imei,
    p.model,
    p.cond,
    l.x,
    l.y,
    st.first_name,
    st.last_name
FROM phone_storage ps
JOIN phones p ON ps.pid = p.pid
JOIN locations l ON ps.lid = l.lid
JOIN students st ON p.sid = st.sid
WHERE ps.retrieved_at IS NULL;

CREATE OR REPLACE VIEW v_empty_locations AS
SELECT
    l.lid,
    l.x,
    l.y
FROM locations l
LEFT JOIN phone_storage ps ON l.lid = ps.lid AND ps.retrieved_at IS NULL
WHERE ps.lid IS NULL;

-- ============================
-- HELPER FUNCTIONS

CREATE OR REPLACE FUNCTION get_phone_storage_history(phone_pid UUID, months_back INT DEFAULT 1)
RETURNS TABLE(
    storage_id INT,
    lid INT,
    x SMALLINT,
    y SMALLINT,
    stored_at TIMESTAMPTZ,
    retrieved_at TIMESTAMPTZ,
    duration INTERVAL
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        ps.id,
        ps.lid,
        l.x,
        l.y,
        ps.stored_at,
        ps.retrieved_at,
        COALESCE(ps.retrieved_at, NOW()) - ps.stored_at as duration
    FROM phone_storage ps
    JOIN locations l ON ps.lid = l.lid
    WHERE ps.pid = phone_pid
      AND ps.stored_at > NOW() - (months_back || ' months')::INTERVAL
    ORDER BY ps.stored_at DESC;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION get_location_usage_history(location_lid INT, months_back INT DEFAULT 1)
RETURNS TABLE(
    phone_pid UUID,
    imei VARCHAR,
    student_sid CHAR,
    stored_at TIMESTAMPTZ,
    retrieved_at TIMESTAMPTZ,
    duration INTERVAL
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        ps.pid,
        p.imei,
        p.sid,
        ps.stored_at,
        ps.retrieved_at,
        COALESCE(ps.retrieved_at, NOW()) - ps.stored_at as duration
    FROM phone_storage ps
    JOIN phones p ON ps.pid = p.pid
    WHERE ps.lid = location_lid
      AND ps.stored_at > NOW() - (months_back || ' months')::INTERVAL
    ORDER BY ps.stored_at DESC;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION get_available_locations(limit_count INT DEFAULT 10)
RETURNS TABLE(
    lid INT,
    x SMALLINT,
    y SMALLINT
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        l.lid,
        l.x,
        l.y
    FROM locations l
    LEFT JOIN phone_storage ps ON l.lid = ps.lid AND ps.retrieved_at IS NULL
    WHERE ps.lid IS NULL
    ORDER BY l.lid
    LIMIT limit_count;
END;
$$ LANGUAGE plpgsql;

COMMENT ON TABLE phone_storage IS 'Junction table tracking which phone is in which location (1 month retention after retrieval). LID auto-assigned if NULL.';

COMMENT ON TABLE slot_baselines IS 'Visual baseline embeddings for each storage slot (runtime state tracking in application only)';

COMMENT ON COLUMN phone_storage.lid IS 'Location ID - auto-assigned to first empty slot if NULL on insert';

COMMENT ON INDEX uniq_active_location IS 'Prevents putting two phones in same location';

COMMENT ON FUNCTION auto_assign_location() IS 'Auto-assigns first available empty location if lid is NULL on phone_storage insert';

-- Evidence tables for admin resolution session proof.
--
-- evidence_sessions: one row per admin session
-- evidence_items:    one row per captured event (photo + metadata)
--
-- Deletion policy (enforced in application, not DB):
--   kept=FALSE rows are deleted on clean session close.
--   kept=TRUE  rows are permanent — no application code deletes them.
-- ============================================================

CREATE TABLE IF NOT EXISTS evidence_sessions (
    id          SERIAL          PRIMARY KEY,
    session_id  VARCHAR(8)      NOT NULL UNIQUE,
    opened_at   TIMESTAMP       NOT NULL,
    closed_at   TIMESTAMP,

    -- 'clean'    → all mismatches resolved, no flagged events, no warnings
    -- 'flagged'  → no_qr_found or declare_missing occurred, or warnings at close
    -- 'incomplete' → session never formally closed (crash / timeout)
    outcome     VARCHAR(20),

    warnings    JSONB,          -- list of warning strings from session close
    kept        BOOLEAN         NOT NULL DEFAULT FALSE
);

CREATE INDEX IF NOT EXISTS idx_evidence_sessions_kept
    ON evidence_sessions (kept)
    WHERE kept = TRUE;

-- ──────────────────────────────────────────────────────────────────────────────

CREATE TABLE IF NOT EXISTS evidence_items (
    id          SERIAL          PRIMARY KEY,
    session_id  VARCHAR(8)      NOT NULL
                                REFERENCES evidence_sessions (session_id)
                                ON DELETE CASCADE,
    captured_at TIMESTAMP       NOT NULL DEFAULT NOW(),

    -- Event that triggered this capture.
    -- Values: 'session_start', 'remove_phone', 'qr_scanned', 'no_qr_found',
    --         'stage_phone', 'unstage_phone', 'place_phone',
    --         'declare_missing', 'needs_deposit', 'session_close'
    event_type  VARCHAR(50)     NOT NULL,

    lid         INTEGER,        -- slot involved (NULL if not applicable)
    pid         VARCHAR(50),    -- phone PID (NULL if not yet identified)
    photo_path  VARCHAR(500),   -- absolute or relative path to JPEG file
                                -- NULL if camera capture failed
    metadata    JSONB           -- event-specific data (expected_lid, blocking_pid, etc.)
);

CREATE INDEX IF NOT EXISTS idx_evidence_items_session
    ON evidence_items (session_id);

CREATE INDEX IF NOT EXISTS idx_evidence_items_event
    ON evidence_items (event_type);
